{ PT_S3_OvernightBiasORB  -  Matteo's overnight-bias opening-range breakout, long AND short.
                                                                               Rules: SPEC.md, S3.
  CHART : NQ or MNQ, 1 MINUTE, 24-hour CME Globex session, Time Zone = Exchange (Central).
  15-minute bars and their ADX are built here from the 1-minute bars (clock-aligned :00/:15/:30/:45);
  the signal is evaluated on the 1-minute bar that closes a 15-minute bar. Daily ATR is built from
  the 17:00-16:00 sessions - no Data2 needed.

  Defaults = the interview (as coded on his chart):
    overnight range 00:00-08:30; the 08:30 open in the TOP third -> longs only, BOTTOM third ->
    shorts only, middle -> no trade. Opening range = 08:30-08:45. A 15-min bar closing above the
    OR high (long bias) / below the OR low (short bias) with ADX(14, 15-min) > 20 -> enter next bar.
    First signal bar closes 09:00, last 14:15. Stop = 0.30 x ATR(15 sessions), target = 3 x stop,
    out at 14:30. One trade a day.
  Variants:  S3-v1 ONStart = 1700      S3-v2 TargetR = 1.5
  Sensitivity: ONStart = 2300 (whiteboard), FirstSignal = 915 or 930                            }

Inputs:
    ONStart(0), ORMins(15),
    FirstSignal(900), LastSignal(1415), TimeExit(1430),
    ADXLen(14), ADXMin(20),
    StopATR(0.30), ATRLen(15), TargetR(3.0), SigMins(15);

Vars:
    ONStartMin(0), FirstMin(0), LastMin(0), ExitMin(0),
    CloseMin(0), OpenMin(0), CloseJD(0), OpenJD(0), DayKey(0), SessKey(0),
    NewDay(false), NewSess(false),
    SessHi(0), SessLo(0), SessCl(0), LastSessCl(0), TRd(0), TRIdx(0), TRCount(0), SumTR(0), ATRd(0),
    InON(false), InONPrev(false), ONH(0), ONL(0), ONValid(false),
    BiasDone(false), DayBias(0), RangePos(0), ORH(0), ORL(0), ORBars(0), TradedToday(false),
    BKey(0), bH(0), bL(0), bC(0), BClosed(true), NewB(false),
    pH(0), pL(0), pC(0), NB(0),
    TRb(0), UpM(0), DnM(0), DMp(0), DMm(0), sTRv(0), sDMp(0), sDMm(0), DIp(0), DIm(0), DXv(0), ADX15(0),
    StopPts(0), k(0);
Arrays:
    TRBuf[100](0);

once begin
    if BarType <> 1 or BarInterval <> 1 then RaiseRunTimeError("PT_S3 needs a 1-minute chart");
    ONStartMin = IntPortion(ONStart / 100) * 60 + Mod(ONStart, 100);
    FirstMin   = IntPortion(FirstSignal / 100) * 60 + Mod(FirstSignal, 100);
    LastMin    = IntPortion(LastSignal / 100) * 60 + Mod(LastSignal, 100);
    ExitMin    = IntPortion(TimeExit / 100) * 60 + Mod(TimeExit, 100);
end;

{ ---- clock ---- }
CloseMin = IntPortion(Time / 100) * 60 + Mod(Time, 100);
OpenMin  = Mod(CloseMin - 1 + 1440, 1440);
CloseJD  = DateToJulian(Date);
if CloseMin = 0 then OpenJD = CloseJD - 1 else OpenJD = CloseJD;
DayKey = OpenJD;
if OpenMin >= 1020 then SessKey = OpenJD + 1 else SessKey = OpenJD;
NewDay  = (CurrentBar = 1) or (DayKey <> DayKey[1]);
NewSess = (CurrentBar = 1) or (SessKey <> SessKey[1]);

{ ---- ATR of the last ATRLen completed sessions ---- }
if NewSess and CurrentBar > 1 then begin
    if LastSessCl > 0 then TRd = MaxList(SessHi, LastSessCl) - MinList(SessLo, LastSessCl)
    else TRd = SessHi - SessLo;
    LastSessCl = SessCl;
    TRBuf[TRIdx] = TRd;
    TRIdx = Mod(TRIdx + 1, ATRLen);
    if TRCount < ATRLen then TRCount = TRCount + 1;
    if TRCount >= ATRLen then begin
        SumTR = 0;
        for k = 0 to ATRLen - 1 begin SumTR = SumTR + TRBuf[k]; end;
        ATRd = SumTR / ATRLen;
    end;
end;
if NewSess then begin SessHi = High; SessLo = Low; end
else begin SessHi = MaxList(SessHi, High); SessLo = MinList(SessLo, Low); end;
SessCl = Close;

if NewDay then begin
    BiasDone = false;
    DayBias = 0;
    ORBars = 0;
    TradedToday = false;
end;

{ ---- overnight range: ONStart (previous evening if > 0) up to 08:29 ---- }
if ONStartMin = 0 then InON = (OpenMin < 510)
else InON = (OpenMin >= ONStartMin) or (OpenMin < 510);
if InON and not InONPrev then begin
    ONH = High;
    ONL = Low;
    ONValid = true;
end else if InON then begin
    ONH = MaxList(ONH, High);
    ONL = MinList(ONL, Low);
end;
InONPrev = InON;

{ ---- bias from the 08:30 open (not on Sunday's 17:00 open, which starts a calendar day too) ---- }
if OpenMin >= 510 and OpenMin < 1020 and not BiasDone then begin
    BiasDone = true;
    DayBias = 0;
    if ONValid and OpenMin <= 515 and ONH > ONL then begin
        RangePos = (Open - ONL) / (ONH - ONL);
        if RangePos >= 2 / 3 then DayBias = 1
        else if RangePos <= 1 / 3 then DayBias = -1;
    end;
    ONValid = false;
end;

{ ---- opening range 08:30 to 08:30 + ORMins ---- }
if OpenMin >= 510 and OpenMin < 510 + ORMins then begin
    if ORBars = 0 then begin ORH = High; ORL = Low; end
    else begin ORH = MaxList(ORH, High); ORL = MinList(ORL, Low); end;
    ORBars = ORBars + 1;
end;

{ ---- 15-minute bars and their Wilder ADX ---- }
BKey = OpenJD * 1440 + OpenMin - Mod(OpenMin, SigMins);
NewB = (CurrentBar = 1) or (BKey <> BKey[1]);
if NewB and CurrentBar > 1 and not BClosed then begin
    { previous bucket ended without a bar on its boundary: close it now }
    if NB > 0 then begin
        TRb = MaxList(bH, pC) - MinList(bL, pC);
        UpM = bH - pH; DnM = pL - bL; DMp = 0; DMm = 0;
        if UpM > DnM and UpM > 0 then DMp = UpM;
        if DnM > UpM and DnM > 0 then DMm = DnM;
        if NB = 1 then begin sTRv = TRb; sDMp = DMp; sDMm = DMm; end
        else begin
            sTRv = sTRv + (TRb - sTRv) / ADXLen;
            sDMp = sDMp + (DMp - sDMp) / ADXLen;
            sDMm = sDMm + (DMm - sDMm) / ADXLen;
        end;
        if sTRv > 0 then begin DIp = 100 * sDMp / sTRv; DIm = 100 * sDMm / sTRv; end
        else begin DIp = 0; DIm = 0; end;
        if DIp + DIm > 0 then DXv = 100 * AbsValue(DIp - DIm) / (DIp + DIm) else DXv = 0;
        if NB = 1 then ADX15 = DXv else ADX15 = ADX15 + (DXv - ADX15) / ADXLen;
    end;
    pH = bH; pL = bL; pC = bC; NB = NB + 1;
    BClosed = true;
end;
if NewB then begin bH = High; bL = Low; BClosed = false; end
else begin bH = MaxList(bH, High); bL = MinList(bL, Low); end;
bC = Close;
if Mod(CloseMin, SigMins) = 0 then begin
    if NB > 0 then begin
        TRb = MaxList(bH, pC) - MinList(bL, pC);
        UpM = bH - pH; DnM = pL - bL; DMp = 0; DMm = 0;
        if UpM > DnM and UpM > 0 then DMp = UpM;
        if DnM > UpM and DnM > 0 then DMm = DnM;
        if NB = 1 then begin sTRv = TRb; sDMp = DMp; sDMm = DMm; end
        else begin
            sTRv = sTRv + (TRb - sTRv) / ADXLen;
            sDMp = sDMp + (DMp - sDMp) / ADXLen;
            sDMm = sDMm + (DMm - sDMm) / ADXLen;
        end;
        if sTRv > 0 then begin DIp = 100 * sDMp / sTRv; DIm = 100 * sDMm / sTRv; end
        else begin DIp = 0; DIm = 0; end;
        if DIp + DIm > 0 then DXv = 100 * AbsValue(DIp - DIm) / (DIp + DIm) else DXv = 0;
        if NB = 1 then ADX15 = DXv else ADX15 = ADX15 + (DXv - ADX15) / ADXLen;
    end;
    pH = bH; pL = bL; pC = bC; NB = NB + 1;
    BClosed = true;

    { ---- signal on the 15-minute close ---- }
    if CloseMin >= FirstMin and CloseMin <= LastMin and MarketPosition = 0 and not TradedToday
       and DayBias <> 0 and ORBars > 0 and ATRd > 0 and NB > 2 and ADX15 > ADXMin then begin
        if DayBias = 1 and Close > ORH then begin
            Buy ("S3 Long") next bar at market;
            TradedToday = true;
        end else if DayBias = -1 and Close < ORL then begin
            SellShort ("S3 Short") next bar at market;
            TradedToday = true;
        end;
    end;
end;

{ ---- exits: stop = StopATR x ATR, target = TargetR x stop, per contract in points ---- }
StopPts = StopATR * ATRd;
SetStopContract;
if StopPts > 0 then begin
    SetStopLoss(StopPts * BigPointValue);
    SetProfitTarget(TargetR * StopPts * BigPointValue);
end;
if CloseMin >= ExitMin and CloseMin <= 960 then begin
    if MarketPosition = 1 then Sell ("S3 Time L") next bar at market;
    if MarketPosition = -1 then BuyToCover ("S3 Time S") next bar at market;
end;
